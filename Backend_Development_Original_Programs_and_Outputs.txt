     [HttpGet]
     public List<Student> Get()
     {
        return students;
     }
     [HttpPost]
     public Student Post(Student student)
     {
        students.Add(student);
        return student;
     }
     [HttpPut]
     public Student Put(Student student)
     {
        var s = students.First(x => x.Id == student.Id);
        s.Name = student.Name;
        return s;
     }
     [HttpDelete]
     public string Delete(int id)
     {
        students.RemoveAll(x => x.Id == id);
        return "Deleted";
     }
  }

using Microsoft.AspNetCore.Authentication.JwtBearer;
using Microsoft.IdentityModel.Tokens;
using System.IdentityModel.Tokens.Jwt;
using System.Security.Claims;
using System.Text;
var b = WebApplication.CreateBuilder(args);
var k = "123456789012345678901234567890123456";
b.Services.AddAuthentication(JwtBearerDefaults.AuthenticationScheme)
.AddJwtBearer(o => o.TokenValidationParameters = new()
{
   ValidateIssuerSigningKey = true,
   IssuerSigningKey = new SymmetricSecurityKey(Encoding.UTF8.GetBytes(k)),
   ValidateIssuer = false,
   ValidateAudience = false
});

b.Services.AddAuthorization();
var app = b.Build();

app.UseAuthentication();
app.UseAuthorization();

app.MapGet("/login", () =>
{
  var t = new JwtSecurityToken(
    claims: new[] { new Claim(ClaimTypes.Name, "Ravi") },
    expires: DateTime.UtcNow.AddMinutes(30),
    signingCredentials: new SigningCredentials(
        new SymmetricSecurityKey(Encoding.UTF8.GetBytes(k)),
        SecurityAlgorithms.HmacSha256));

   return new { token = new JwtSecurityTokenHandler().WriteToken(t) };
});

app.MapGet("/secure", () => "Welcome to Secure API!")
  .RequireAuthorization();

app.Run();

PRACTICAL 13: Implement Role-Based Authorization for Admin and User Roles
==============================================================================
ORIGINAL PROGRAM / CODE (extracted from the practical record)

Program.cs
using Microsoft.AspNetCore.Authentication.JwtBearer;
using Microsoft.IdentityModel.Tokens;
using System.IdentityModel.Tokens.Jwt;
using System.Security.Claims;
using System.Text;
var b = WebApplication.CreateBuilder(args);
var k = "123456789012345678901234567890123456";
b.Services.AddAuthentication(JwtBearerDefaults.AuthenticationScheme)
.AddJwtBearer(o => o.TokenValidationParameters = new()
{
   ValidateIssuerSigningKey = true,
   IssuerSigningKey = new SymmetricSecurityKey(Encoding.UTF8.GetBytes(k)),
   ValidateIssuer = false,
   ValidateAudience = false
});
b.Services.AddAuthorization();
var app = b.Build();
app.UseAuthentication();
app.UseAuthorization();
app.MapGet("/login/admin", () => Token("Admin"));
app.MapGet("/login/user", () => Token("User"));
app.MapGet("/admin", () => "Welcome Admin!")
  .RequireAuthorization(x => x.RequireRole("Admin"));

app.MapGet("/user", () => "Welcome User!")
  .RequireAuthorization(x => x.RequireRole("Admin", "User"));
app.Run();
string Token(string role)
{
   var t = new JwtSecurityToken(
      claims: new[] { new Claim(ClaimTypes.Role, role) },
      expires: DateTime.UtcNow.AddMinutes(30),
      signingCredentials: new SigningCredentials(
         new SymmetricSecurityKey(Encoding.UTF8.GetBytes(k)),
         SecurityAlgorithms.HmacSha256));
   return new JwtSecurityTokenHandler().WriteToken(t);
}
